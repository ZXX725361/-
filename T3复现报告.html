1. 研究问题

软件程序越来越多地被企业用于定价。本文的核心问题是：定价算法能否"自发地"学会合谋？作者构建了基于Q-learning（强化学习的一种）的定价智能体，让它们在重复博弈的模拟市场中反复互动，考察算法是否会在没有任何显式沟通或人工指令的情况下，收敛到超竞争价格水平。

核心发现：Q-learning算法确实会系统地学习到合谋策略——在对称双寡头基准下，算法最终将价格定在接近垄断水平，且这种合谋由"惩罚偏离者"的策略机制维持。

2. 模型与参数设定

经济环境为无限期重复差异化产品Bertrand博弈，需求采用Logit形式。每个企业使用Q-learning算法更新定价策略，采用ε-greedy探索策略。

6. 惩罚机制分析（脉冲响应）

论文的核心发现之一是：合谋价格不是简单的静态高价，而是由惩罚-回报机制维持的。为验证这一点，在算法收敛后，强制企业1在t=1时偏离到短期最优降价，观察双方后续价格反应。

机制解读

阶段时间行为
合作期t=0双方定价1.89（接近垄断价格1.925）
偏离t=1企业1降价至1.66（短期最优反应）
惩罚开始t=2–4双方大幅降价至Nash价格附近（1.43–1.47）
惩罚谷底t=5偏离企业降至最低价1.43
回归合作t=6–9双方逐步提价，经过4–5期过渡期
恢复合作t≥9回到合作价格1.89
关键发现：惩罚策略并非"冷酷触发"（grim trigger），而是论文所说的"胡萝卜加大棒"（stick-and-carrot）模式——有限期惩罚后逐渐回到合作价格。这与Abreu (1986)的极值均衡理论一致，但惩罚不如理论上那么严厉（价格在惩罚期仍高于Nash水平）。

7. 复现结论

1. 合谋确实自发涌现：Q-learning算法在没有任何人工指令、没有沟通的情况下，系统性地学习到超竞争定价策略，平均实现81%的潜在合谋利润。

2. 策略是合谋性的而非误优化：偏离均衡价格会引发对方的惩罚性降价，且惩罚期后双方回到合作——这是真正的奖励-惩罚机制，而非算法未能最优定价。

3. 与论文高度一致：利润增益、收敛时间、惩罚模式均与Calvano et al. (2020)报告的结果吻合，验证了论文结论的可复现性。

Calvano et al. (2020) 完整复现 - 带可视化

import numpy as np
from itertools import product
import time
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# 设置中文字体
plt.rcParams['font.sans-serif'] = ['DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

class BertrandLogitGame:
    def __init__(self, n=2, c=None, a=None, a0=0.0, mu=0.25, delta=0.95):
        self.n = n
        self.c = np.array(c if c is not None else [1.0] * n)
        self.a = np.array(a if a is not None else [2.0] * n)
        self.a0 = a0
        self.mu = mu
        self.delta = delta

    def all_profits(self, prices):
        prices = np.array(prices, dtype=float)
        v = np.exp((self.a - prices) / self.mu)
        v0 = np.exp(self.a0 / self.mu)
        q = v / (v.sum() + v0)
        return (prices - self.c) * q


def run_session(game, price_grid, alpha=0.15, beta=4e-6,
                max_periods=1500000, convergence_window=100000,
                check_interval=5000, seed=42, verbose=True):
    """运行单个Q-learning session"""
    np.random.seed(seed)
    n = game.n
    m = len(price_grid)
    n_states = m ** n

    # 预计算所有组合利润
    p_indices = np.array(list(product(range(m), repeat=n)))
    prices_all = price_grid[p_indices]
    v = np.exp((game.a[np.newaxis, :] - prices_all) / game.mu)
    v0 = np.exp(game.a0 / game.mu)
    q_all = v / (v.sum(axis=1, keepdims=True) + v0)
    profits_all = (prices_all - game.c[np.newaxis, :]) * q_all

    # Q表初始化
    Q = np.zeros((n, n_states, m))
    for firm in range(n):
        for ai in range(m):
            mask = p_indices[:, firm] == ai
            avg_profit = profits_all[mask, firm].mean()
            Q[firm, :, ai] = avg_profit / (1 - game.delta)

    state = np.random.randint(n_states)
    profits_hist = []
    prices_hist = []
    eps_hist = []

    prev_best = None
    consecutive_same = 0
    converged = False
    conv_period = max_periods

    t_start = time.time()

    for t in range(max_periods):
        eps = np.exp(-beta * t)

        # 选择动作
        actions = np.zeros(n, dtype=int)
        for firm in range(n):
            if np.random.random() < eps:
                actions[firm] = np.random.randint(m)
            else:
                qv = Q[firm, state, :]
                best = np.where(qv == np.max(qv))[0]
                actions[firm] = np.random.choice(best)

        # 利润
        comb = 0
        for i in range(n):
            comb = comb * m + actions[i]
        cur_profits = profits_all[comb, :]
        cur_prices = prices_all[comb, :]
        next_state = comb

        # 更新Q
        for firm in range(n):
            best_next = np.max(Q[firm, next_state, :])
            td_target = cur_profits[firm] + game.delta * best_next
            Q[firm, state, actions[firm]] += alpha * (td_target - Q[firm, state, actions[firm]])

        profits_hist.append(cur_profits.mean())
        prices_hist.append(cur_prices.copy())
        eps_hist.append(eps)
        state = next_state

        # 收敛检查
        if (t + 1) % check_interval == 0:
            cur_best = np.argmax(Q, axis=2)
            if prev_best is not None and np.array_equal(cur_best, prev_best):
                consecutive_same += check_interval
            else:
                consecutive_same = 0
            prev_best = cur_best.copy()
            if consecutive_same >= convergence_window:
                converged = True
                conv_period = t + 1
                break

        if verbose and (t + 1) % 200000 == 0:
            elapsed = time.time() - t_start
            p_recent = np.mean(prices_hist[-1000:], axis=0)
            print(f"    t={t+1:>8d}: p̄={p_recent.mean():.4f}, ε={eps:.4f}, {elapsed:.1f}s")

    elapsed = time.time() - t_start
    if verbose:
        print(f"    完成: conv={converged}, t={conv_period}, {elapsed:.1f}s")

    return {
        'Q': Q, 'converged': converged, 'conv_period': conv_period,
        'prices': np.array(prices_hist), 'profits': np.array(profits_hist),
        'eps': np.array(eps_hist), 'elapsed': elapsed
    }


def impulse_response(game, price_grid, Q, T=20):
    """脉冲响应分析"""
    n = game.n
    m = len(price_grid)
    strategies = [np.argmax(Q[f], axis=1) for f in range(n)]

    # 找吸收态
    state = np.random.randint(m**n)
    for _ in range(1000):
        acts = [strategies[f][state] for f in range(n)]
        comb = 0
        for i in range(n):
            comb = comb * m + acts[i]
        state = comb
    coop_actions = tuple(acts)
    coop_price = price_grid[coop_actions[0]]

    # 找企业1对合作价格p2的最优反应
    p2 = coop_price
    p_fine = np.linspace(price_grid[0], price_grid[-1], 1000)
    best_dev_idx = 0
    best_dev_profit = -np.inf
    for i, p1 in enumerate(p_fine):
        pr = game.all_profits([p1, p2])[0]
        if pr > best_dev_profit:
            best_dev_profit = pr
            best_dev_idx = i
    best_dev_price = p_fine[best_dev_idx]
    dev_idx = np.argmin(np.abs(price_grid - best_dev_price))

    # 模拟脉冲响应
    coop_state = 0
    for i in range(n):
        coop_state = coop_state * m + coop_actions[i]
    state = coop_state

    paths = np.zeros((T, n))
    for t in range(T):
        if t == 1:
            acts = [dev_idx, strategies[1][state]]
        else:
            acts = [strategies[f][state] for f in range(n)]
        paths[t] = price_grid[acts]
        comb = 0
        for i in range(n):
            comb = comb * m + acts[i]
        state = comb

    return paths, coop_price, price_grid[dev_idx]

    game = BertrandLogitGame(n=2, c=[1.,1.], a=[2.,2.], a0=0., mu=0.25, delta=0.95)

    # 精确均衡价格
    from scipy.optimize import brentq, minimize_scalar

        res = minimize_scalar(lambda p1: -game.all_profits([p1,p2])[0],

        return res.x
    p_nash = brentq(lambda p: br(p)-p, 1.0, 3.0)
    p_mon = minimize_scalar(lambda p: -game.all_profits([p,p]).sum(),

    pi_mon = game.all_profits([p_mon,p_mon])[0]

    grid = np.linspace(p_lo, p_hi, m)
    print(f"价格网格: [{p_lo:.4f}, {p_hi:.4f}], m={m}")

    # 运行session
